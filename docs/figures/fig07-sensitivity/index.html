Figure 7 · Sensitivity to the bulk configuration

Leipzig Institute for Meteorology (LIM), Leipzig University

Paper: Sect. 4.2 · Script: figures_manuscript/fig07_sensitivity.py · Method: Chapter 5

Construction¶

Tornado-chart layout: three shared-y panels, (a) regression slope b on the EC reference (ideal 1), (b) correlation r, (c) bias (ideal 0). Each configuration row shows the three offline bulk estimates as dots (T-Bird green, ERA5-MODIS orange, ERA5-only blue; no connector line); the model-diagnosed ERA5/CARRA fluxes are the dashed vertical benchmark lines. Representative configurations only (full matrix: Table C1).

Data: obs_/rs_ metrics from sensitivity_summary_table.csv; the ERA5-only bulk is the rs_* pathway of the matching ‘_era5only’ run; diag metrics are computed from sensitivity_figure_data.nc against the EC reference.

Reproduction¶

pixi run python figures_manuscript/fig07_sensitivity.py

The script writes figures_manuscript/output/fig07_sensitivity.pdf and .png at the printed width (one column, 8.3 cm); pixi run figures writes all figures. It uses fluxpy.figstyle.

Source code (figures_manuscript/fig07_sensitivity.py)
"""Fig. 7 (fig:sensitivity_summary) — sensitivity in the slope/r framework (manuscript Sect. 4.2).

Tornado-chart layout: three shared-y panels, (a) regression slope b on the EC
reference (ideal 1), (b) correlation r, (c) bias (ideal 0). Each configuration
row shows the three offline bulk estimates as dots (T-Bird green, ERA5-MODIS
orange, ERA5-only blue; no connector line); the model-diagnosed
ERA5/CARRA fluxes are the dashed vertical benchmark lines. Representative
configurations only (full matrix: Table C1).

Data: obs_*/rs_* metrics from sensitivity_summary_table.csv; the ERA5-only
bulk is the rs_* pathway of the matching '<label>_era5only' run; diag metrics
are computed from sensitivity_figure_data.nc against the EC reference.

Run:  pixi run python figures_manuscript/fig07_sensitivity.py
"""
import sys
from pathlib import Path

import numpy as np
import pandas as pd
import xarray as xr
from matplotlib.lines import Line2D

REPO = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(REPO))
from fluxpy import figstyle as fs  # noqa: E402

fs.apply()

df = pd.read_csv(REPO / "data/model_output/sensitivity_summary_table.csv").set_index("label")
FIG = xr.open_dataset(REPO / "data/model_output/sensitivity_figure_data.nc")
EC = np.asarray(FIG["H_obs_ec"])


def diag_metrics(key):
    """(slope, r, bias) of a model-diagnosed flux against the EC reference."""
    h = np.asarray(FIG[key])
    m = np.isfinite(h) & np.isfinite(EC)
    return (float(np.polyfit(EC[m], h[m], 1)[0]),
            float(np.corrcoef(EC[m], h[m])[0, 1]),
            float(np.mean(h[m] - EC[m])))


# ("header", text) rows get no tick and a thin separator line above them;
# ("diag", label, var) rows carry ONE dot — the model-diagnosed fluxes do not
# depend on the bulk configuration.
ROWS = [
    ("row", "Reference ($\\alpha_T = 10^{-1}$)", "GLGS20_ref"),
    ("row", "$\\alpha_T = 10^{-2}$ (amplitude-matched)", "GLGS20_ztr1e-02"),
    ("header", "Stability scheme"),
    ("row", "Businger–Dyer", "BD_ref"),
    ("row", "SHEBA (GAFGP07)", "GAFGP07_ref"),
    ("row", "ZE07-I", "ZE07-I_ref"),
    ("header", "Momentum roughness"),
    ("row", "MOSAiC set", "GLGS20_mosaic"),
    ("row", "Schnierstein set", "GLGS20_nschnier"),
    ("row", "$4{\\times}10^{-4}$ m uniform", "GLGS20_z0c4e-04"),
    ("header", "Model-diagnosed"),
    ("diag", "ERA5", "H_era_diag"),
    ("diag", "CARRA$^{*}$", "H_car_diag"),
]

PANELS = [("slope", "Slope $b$", (0.0, 2.4), [0, 1, 2], 1.0),
          ("r", "Corr. $r$", (0.25, 0.80), [0.3, 0.7], None),
          ("bias", "Bias\n(W$\\,$m$^{-2}$)", (-4.6, 4.6), [-4, 0, 4], 0.0)]
DIAG = {k: dict(zip(("slope", "r", "bias"), diag_metrics(k)))
        for k in ("H_era_diag", "H_car_diag")}


def row_values(label, metric):
    """(T-Bird, ERA5-MODIS, ERA5-only) value triple for one configuration."""
    return (df.loc[label, f"obs_{metric}"],
            df.loc[label, f"rs_{metric}"],
            df.loc[f"{label}_era5only", f"rs_{metric}"])


fig, axes = fs.figure(width=fs.COL_W, aspect=0.95, ncols=3, sharey=True)
ypos = {}                      # y position of every entry, top to bottom
for i, entry in enumerate(ROWS):
    ypos[i] = -float(i)
tick_rows = [(ypos[i], e[1]) for i, e in enumerate(ROWS) if e[0] != "header"]
headers = [(ypos[i], e[1]) for i, e in enumerate(ROWS) if e[0] == "header"]

for ax, (metric, xlab, xlim, ticks, ideal) in zip(axes, PANELS):
    for j, (yy, _) in enumerate(tick_rows):               # zebra banding
        if j % 2 == 0:
            ax.axhspan(yy - 0.5, yy + 0.5, color="0.94", zorder=0)
    for yy, _ in headers:                                 # thin section separators
        ax.axhline(yy + 0.5, color="0.75", lw=0.5, zorder=1)
    if ideal is not None:
        ax.axvline(ideal, color="0.35", lw=0.8, ls="--", zorder=1)
    for i, entry in enumerate(ROWS):
        yy = ypos[i]
        if entry[0] == "row":
            vo, vr, ve = row_values(entry[2], metric)
            # no connector between the three dots: there are no values between them
            for v, key in [(vo, "H_obs_bulk"), (vr, "H_rs_bulk"), (ve, "H_era_bulk")]:
                ax.plot(v, yy, "o", ms=4.5, color=fs.C[key], mec="white", mew=0.5,
                        zorder=4)
        elif entry[0] == "diag":                          # one dot, no configuration
            ax.plot(DIAG[entry[2]][metric], yy, "o", ms=4.5, color=fs.C[entry[2]],
                    mec="white", mew=0.5, zorder=4)
    ax.set_xlim(xlim)
    ax.set_xticks(ticks)
    ax.set_xlabel(xlab)
    ax.set_ylim(min(ypos.values()) - 0.6, 0.6)
    ax.tick_params(axis="y", length=0)

axes[0].set_yticks([yy for yy, _ in tick_rows])
axes[0].set_yticklabels([lbl for _, lbl in tick_rows])
axes[0].get_yticklabels()[0].set_fontweight("bold")
for yy, text in headers:                                  # tickless header labels
    axes[0].text(-0.06, yy, text, transform=axes[0].get_yaxis_transform(),
                 ha="right", va="center", fontstyle="italic", color="0.35")

fig.legend(handles=[
    Line2D([0], [0], marker="o", lw=0, color=fs.C["H_obs_bulk"], ms=4.5,
           mec="white", mew=0.5, label=fs.L["H_obs_bulk"]),
    Line2D([0], [0], marker="o", lw=0, color=fs.C["H_rs_bulk"], ms=4.5,
           mec="white", mew=0.5, label=fs.L["H_rs_bulk"]),
    Line2D([0], [0], marker="o", lw=0, color=fs.C["H_era_bulk"], ms=4.5,
           mec="white", mew=0.5, label=fs.L["H_era_bulk"]),
    Line2D([0], [0], marker="o", lw=0, color=fs.C["H_era_diag"], ms=4.5,
           mec="white", mew=0.5, label=fs.L["H_era_diag"]),
    Line2D([0], [0], marker="o", lw=0, color=fs.C["H_car_diag"], ms=4.5,
           mec="white", mew=0.5, label=fs.L["H_car_diag"])],
    loc="outside upper center", ncol=3, columnspacing=0.8, handlelength=1.2,
    handletextpad=0.4)

for ax, lbl in zip(axes, ["(a)", "(b)", "(c)"]):
    fs.panel_label(ax, lbl, x=0.06)

fs.save(fig, "fig07_sensitivity")
