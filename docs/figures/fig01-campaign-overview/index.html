Figure 1 · Campaign overview

Leipzig Institute for Meteorology (LIM), Leipzig University

Paper: Sect. 2.1 · Script: figures_manuscript/fig01_campaign_overview.py · Method: Chapter 1

Construction¶

April-2024 mean sea-ice concentration (merged MODIS-AMSR2, 1 km) with all 13 Polar 6 master tracks of BACSAM-II (PANGAEA, doi:10.1594/PANGAEA.971763). The four analysed low-level flight days are highlighted in the batlow day colours (17 April in red, drawn on top) and their focus regions (the detail panels of the low-level flight figure) are drawn as labelled rectangles. A flight is coloured only if it overlaps in time with the analysed 5-km segments (three flights flew on 13 April; only the one containing the low-level legs is highlighted).

Reproduction¶

pixi run python figures_manuscript/fig01_campaign_overview.py

The script writes figures_manuscript/output/fig01_campaign_overview.pdf and .png at the printed width (one column, 8.3 cm); pixi run figures writes all figures. It uses fluxpy.figstyle.

Source code (figures_manuscript/fig01_campaign_overview.py)
"""Fig. 1 -- campaign overview map (manuscript Sect. 2.1, fig:campaign_overview).

April-2024 mean sea-ice concentration (merged MODIS-AMSR2, 1 km) with all 13
Polar 6 master tracks of BACSAM-II (PANGAEA, doi:10.1594/PANGAEA.971763).
The four analysed low-level flight days are highlighted in the batlow day
colours (17 April in red, drawn on top) and their focus regions (the detail panels of the low-level flight
figure) are drawn as labelled rectangles.  A flight is coloured only if it
overlaps in time with the analysed 5-km segments (three flights flew on
13 April; only the one containing the low-level legs is highlighted).

Run:  pixi run python figures_manuscript/fig01_campaign_overview.py
"""
import sys
from io import StringIO
from pathlib import Path

REPO = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(REPO))

import matplotlib

matplotlib.use("Agg")

from fluxpy import figstyle as fs

fs.apply()

import numpy as np
import pandas as pd
import xarray as xr
import matplotlib.patheffects as pe
from matplotlib.lines import Line2D
from matplotlib.patches import Rectangle
from cartopy import crs as ccrs
import cartopy.feature as cfeature
from cmocean import cm as cmo
from cmcrameri import cm as cmc

TRACK_DIR = REPO / "data" / "observations" / "raw" / "polar6" / "mastertracks"
SIC_FP = (REPO / "data" / "observations" / "processed" / "satellite_unified"
          / "sic_modis-aqua_amsr2-gcom-w1_merged_nh_1000m_APRIL.nc")
AR_FP = (REPO / "data" / "observations" / "ar"
         / "analysis_ready_polar6_turbulent_fluxes_latest_5km.nc")

LONGYEARBYEN = (15.63, 78.22)
PC = ccrs.PlateCarree()
HALO = [pe.withStroke(linewidth=1.8, foreground="white")]


def read_mastertrack(fp: Path) -> pd.DataFrame:
    """Parse a PANGAEA mastertrack .tab file (skip the /* ... */ header)."""
    txt = fp.read_text(encoding="utf-8", errors="replace")
    df = pd.read_csv(StringIO(txt[txt.find("*/") + 2:]), sep="\t")
    df.columns = ["time", "lat", "lon", "alt"]
    df["time"] = pd.to_datetime(df["time"])
    return df


# -- analysed low-level days, batlow day colours, focus regions --------------
low = xr.open_dataset(AR_FP).pipe(lambda d: d.where(d.altitude <= 50, drop=True))
seg_times = pd.to_datetime(low.start_time.values)
low_alt_days = np.unique(seg_times.strftime("%Y-%m-%d"))
day_color = dict(zip(low_alt_days, cmc.batlow(np.linspace(0, 1, len(low_alt_days)))))
# 17 April: batlow's mid-range olive vanished next to the 23-April tracks
# (co-author round 3) -> a saturated red, drawn above the 23-April tracks.
# Keep in sync with the frame colours of fig04_flux_maps.py.
DAY_COLOR_0417 = "#d7191c"
day_color["2024-04-17"] = DAY_COLOR_0417
DAY_ZORDER = {"2024-04-17": 1}        # extra zorder: 17 April on top
analysed_days = [d for d in low_alt_days if d != "2024-04-25"]

# focus regions of the detail panels (a)-(d) of fig04_flux_maps.py
focus = {}
for panel, day in zip("abcd", analysed_days):
    ds_day = low.sel(start_time=day)
    x, y = ds_day.longitude.values, ds_day.latitude.values
    focus[f"({panel})"] = (day, [np.nanmin(x) - 1.0, np.nanmax(x) + 1.0,
                                 np.nanmin(y) - 0.12, np.nanmax(y) + 0.12])

# -- master tracks: coloured if they overlap the analysed segments -----------
tracks = []
for fp in sorted(TRACK_DIR.glob("*_mastertrack.tab")):
    df = read_mastertrack(fp)
    day = df["time"].dt.strftime("%Y-%m-%d").iloc[0]
    if day in low_alt_days:
        overlaps = True
    else:
        overlaps = False
    tracks.append((df, day, overlaps and day in analysed_days))

# -- April-mean SIC backdrop (25 April excluded: bad data, 100 % everywhere) -
sic = (xr.open_dataset(SIC_FP).squeeze().sortby("time")
       .drop_sel(time="2024-04-25").sic.mean("time"))

# -- figure ------------------------------------------------------------------
fig, ax = fs.figure(width=fs.COL_W, aspect=0.9,
                    subplot_kw={"projection": ccrs.NorthPolarStereo(
                        central_longitude=LONGYEARBYEN[0])})
lat_north = max(86.0, max(df.lat.max() for df, *_ in tracks) + 0.4)
ax.set_extent([-20, 25, 74, lat_north], crs=PC)

im = ax.contourf(sic.lon, sic.lat, sic, transform=PC, cmap=cmo.ice,
                 levels=np.arange(0, 110, 10), alpha=0.6, rasterized=True)
ax.add_feature(cfeature.LAND, facecolor="lightgray", zorder=1)
ax.add_feature(cfeature.COASTLINE, linewidth=0.5, zorder=2)

# graticule as labelled contours of 2-D lat/lon fields (74\N{DEGREE SIGN}N and
# 0\N{DEGREE SIGN} are drawn but not labelled -- their labels clip at the frame)
glon, glat = np.meshgrid(np.linspace(-40, 45, 341), np.linspace(70, 88, 181))
for field, levels, labelled, fmt in [
        (glat, np.arange(72, 86, 2), np.arange(76, 86, 2),
         lambda v: f"{v:.0f}\N{DEGREE SIGN}N"),
        (glon, np.arange(-30, 45, 10), [-30, -20, -10, 10, 20, 30, 40],
         lambda v: f"{abs(v):.0f}\N{DEGREE SIGN}{'E' if v >= 0 else 'W'}")]:
    cs = ax.contour(glon, glat, field, levels=levels, colors="grey",
                    linewidths=0.4, linestyles=":", transform=PC, zorder=3)
    ax.clabel(cs, levels=[v for v in labelled if v in cs.levels], fmt=fmt,
              fontsize=fs.FONT_SIZE - 1, inline=True)

# flight tracks: grey for all, day colours for the analysed flights; the grey
# tracks are dark with a white outline, so that they stay visible over the light
# grey land and the dark open ocean alike
OTHER_TRACK = dict(color="0.3", linewidth=0.8,
                   path_effects=[pe.withStroke(linewidth=1.6, foreground="white")])
for df, day, analysed in tracks:
    if analysed:
        ax.plot(df.lon, df.lat, color=day_color[day], linewidth=1,
                transform=PC, zorder=6 + 0.5 * DAY_ZORDER.get(day, 0),
                path_effects=[pe.withStroke(linewidth=1.1, foreground="white")])
    else:
        ax.plot(df.lon, df.lat, **OTHER_TRACK, transform=PC, zorder=4)

# focus-region rectangles; (c) and (d) almost coincide, so their labels sit at
# opposite corners of the shared box (d: north-west, c: south-east), each tied
# to its corner by a short leader line; (b) sits below its box, inside the map
# frame (above it, it was cut by the frame)
pc_data = PC._as_mpl_transform(ax)
LABEL_POS = {   # (anchor corner of the box, label offset in points, ha, va)
    "(a)": ("top-left", (-3, 0), "right", "top"),
    "(b)": ("bottom-centre", (0, -3), "center", "top"),
    "(c)": ("bottom-right", (6, -5), "left", "top"),
    "(d)": ("top-left", (-7, 6), "right", "bottom"),
}
for name, (day, (x0, x1, y0, y1)) in focus.items():
    ax.add_patch(Rectangle((x0, y0), x1 - x0, y1 - y0, fill=False,
                           linewidth=.75, edgecolor=day_color[day],
                           transform=PC, zorder=7 + 0.5 * DAY_ZORDER.get(day, 0),
                           alpha=.7, linestyle="--"))
    corner, offset, ha, va = LABEL_POS[name]
    xy = {"top-left": (x0, y1), "bottom-right": (x1, y0),
          "bottom-centre": (0.5 * (x0 + x1), y0)}[corner]
    leader = (dict(arrowstyle="-", color=day_color[day], lw=0.6, shrinkA=0, shrinkB=0)
              if name in ("(c)", "(d)") else None)
    ax.annotate(name, xy=xy, xycoords=pc_data, xytext=offset,
                textcoords="offset points", color=day_color[day], weight="bold",
                ha=ha, va=va, zorder=8, path_effects=HALO, arrowprops=leader)

# geographical references
ax.plot(*LONGYEARBYEN, marker="*", markersize=9, color="k",
        transform=PC, zorder=7)
ax.text(LONGYEARBYEN[0] + 1.1, LONGYEARBYEN[1] - 0.72, "LYR",
        fontsize=fs.FONT_SIZE - 1, transform=PC, zorder=8, path_effects=HALO)
for name, lon, lat, col in [("Svalbard", 19.0, 78.9, "dimgrey"),
                            ("Greenland", -19.0, 80.3, "dimgrey"),
                            ("Fram Strait", -4.0, 79.3, "black"),
                            ("Arctic Ocean", 2.0, 85.0, "black")]:
    ax.text(lon, lat, name, fontsize=fs.FONT_SIZE - 1, color=col,
            style="italic", ha="center", va="center", transform=PC, zorder=7,
            path_effects=HALO)

# legend and colourbar
handles = [Line2D([0], [0], color=day_color[d], linewidth=1.8,
                  label=pd.Timestamp(d).strftime("%d %B"))
           for d in analysed_days]
handles.append(Line2D([0], [0], **OTHER_TRACK, label="Other flights"))
ax.legend(handles=handles, loc="lower left", frameon=True, framealpha=0.9,
          title="Flight tracks")

cbar = fig.colorbar(im, ax=ax, fraction=0.035, pad=0.02)
cbar.set_label("Sea-ice concentration (%)")

ax.spines["geo"].set_visible(False)

fs.save(fig, "fig01_campaign_overview")
